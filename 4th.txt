
import numpy as np
import random
import matplotlib.pyplot as plt
import torch
import torch.nn as nn
import torch.optim as optim

from collections import deque

# Create environment
n_states = 5
n_actions = 2

# Create Neural Network
class DQN(nn.Module):
    def __init__(self):
        super().__init__()

        self.fc1 = nn.Linear(n_states, 24)
        self.fc2 = nn.Linear(24, 24)
        self.fc3 = nn.Linear(24, n_actions)

    def forward(self, x):
        x = torch.relu(self.fc1(x))
        x = torch.relu(self.fc2(x))
        return self.fc3(x)


# Create model
model = DQN()
target_model = DQN()

target_model.load_state_dict(model.state_dict())

optimizer = optim.Adam(model.parameters(), lr=0.001)

memory = deque(maxlen=2000)

gamma = 0.95
epsilon = 1.0

rewards_list = []

# Convert state into one-hot format
def get_state(pos):
    state = np.zeros(n_states)
    state[pos] = 1
    return state


# Training
for episode in range(300):

    position = 0
    total_reward = 0
    done = False

    while not done:

        state = get_state(position)

        # Select action
        if random.random() < epsilon:
            action = random.randint(0, 1)
        else:
            with torch.no_grad():
                action = model(
                    torch.FloatTensor(state)
                ).argmax().item()

        # Move left or right
        if action == 0:
            next_position = max(0, position - 1)
        else:
            next_position = min(4, position + 1)

        # Calculate reward
        reward = 10 if next_position == 4 else -1
        done = next_position == 4

        next_state = get_state(next_position)

        # Store experience
        memory.append(
            (state, action, reward, next_state, done)
        )

        position = next_position
        total_reward += reward

        # Train from past experiences
        if len(memory) >= 32:

            batch = random.sample(memory, 32)

            for s, a, r, ns, d in batch:

                s = torch.FloatTensor(s)
                ns = torch.FloatTensor(ns)

                target = r

                if not d:
                    with torch.no_grad():
                        target += gamma * target_model(ns).max().item()

                prediction = model(s)[a]

                loss = (prediction - target) ** 2

                optimizer.zero_grad()
                loss.backward()
                optimizer.step()

    # Reduce random actions
    epsilon = max(0.01, epsilon * 0.98)

    # Update target network
    if episode % 10 == 0:
        target_model.load_state_dict(model.state_dict())

    rewards_list.append(total_reward)


# Display results
plt.figure(figsize=(8, 5))
plt.plot(rewards_list)
plt.xlabel("Episodes")
plt.ylabel("Total Reward")
plt.title("DQN Learning Performance")
plt.show()

print("Training Completed!")
